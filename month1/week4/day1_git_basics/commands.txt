# ==========================
# 🧭 Основные команды Git
# ==========================

# Инициализация нового репозитория
git init

# Проверка состояния (отслеженные / неотслеженные файлы)
git status

# Добавление файлов в staging area
git add <имя_файла>          # добавить один файл
git add .                    # добавить все файлы

# Создание коммита
git commit -m "сообщение"    # пример: git commit -m "feat: add hello.py"

# Просмотр истории коммитов
git log
git log --oneline            # короткий формат
git log --graph --oneline --decorate --all  # визуальное дерево

# Просмотр изменений
git diff                     # показать различия между изменёнными и закоммиченными файлами
git diff <commit1> <commit2> # сравнить разные коммиты

# ==========================
# 🧱 Работа с файлами
# ==========================

# Удаление файла из репозитория
git rm <имя_файла>

# Переименование или перемещение файла
git mv <старое_имя> <новое_имя>

# ==========================
# 🔄 Откат и восстановление
# ==========================

# Отменить изменения в файле (вернуть к последнему коммиту)
git restore <имя_файла>

# Отменить добавление файла в staging
git restore --staged <имя_файла>

# Переключиться на предыдущий коммит
git checkout <commit_id>

# Создать новую ветку из конкретного коммита
git checkout -b <новая_ветка> <commit_id>

# ==========================
# 🧰 Прочие полезные команды
# ==========================

# Создать .gitignore
touch .gitignore

# Проверить текущую конфигурацию пользователя
git config user.name
git config user.email

# Настроить пользователя
git config --global user.name "Your Name"
git config --global user.email "your@email.com"
