# ==========================
# Работа с ветками
# ==========================

# Просмотр всех веток
git branch

# Создание новой ветки
git branch <имя_ветки>
# Пример:
git branch feature/day2

# Переключение на ветку
git checkout <имя_ветки>
# Пример:
git checkout feature/day2

# Создание и переключение на новую ветку одной командой
git checkout -b <имя_ветки>
# Пример:
git checkout -b feature/day2

# Проверка, в какой ветке сейчас находимся
git status

# ==========================
# Слияние веток (merge)
# ==========================

# Переключаемся в основную ветку
git checkout main

# Сливаем изменения из feature-ветки в main
git merge feature/day2

# Разрешаем конфликты (если появились), затем:
git add .
git commit -m "merge: feature/day2 merged into main"

# ==========================
# Подключение к удалённому репозиторию (GitHub)
# ==========================

# Подключить удалённый репозиторий
git remote add origin <URL>
# Пример:
# git remote add origin git@github.com:USERNAME/repo-name.git

# Проверить подключение
git remote -v

# ==========================
# Отправка изменений
# ==========================

# Отправить основную ветку в GitHub (первый раз)
git push -u origin main

# Отправить конкретную ветку
git push -u origin feature/day2

# ==========================
# Получение изменений с GitHub
# ==========================

# Обновить локальный main из удалённого репозитория
git pull origin main

# ==========================
# Удаление веток
# ==========================

# Удалить локальную ветку
git branch -d <имя_ветки>
# Пример:
git branch -d feature/day2

# Удалить ветку на GitHub
git push origin --delete <имя_ветки>
# Пример:
git push origin --delete feature/day2

# ==========================
# Полезные команды
# ==========================

# Показать ветки с последними коммитами
git branch -vv

# Показать историю коммитов в виде дерева
git log --oneline --graph --decorate --all
